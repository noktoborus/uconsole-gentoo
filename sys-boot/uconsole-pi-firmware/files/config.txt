# /boot/config.txt of the uConsole, read by the firmware at boot. From
# the file of the device on 2026-10-19. sys-boot/uconsole-pi-firmware
# installs it; CONFIG_PROTECT keeps the changes on an update.

[pi4]
hdmi_enable_4kp60=1
dtoverlay=clockworkpi-uconsole
dtoverlay=vc4-kms-v3d-pi4,cma-384
enable_uart=1
arm_freq_min=400
arm_freq=1900
gpu_mem=256
over_voltage=5
dtoverlay=audremap,pins_12_13
dtparam=audio=on

[pi5]
dtoverlay=clockworkpi-uconsole-cm5
dtoverlay=vc4-kms-v3d-pi5,cma-384
dtoverlay=uconsole-cm5-sound
dtparam=uart0
dtparam=pciex1
dtparam=pciex1_gen=3
arm_freq_min=500
arm_freq=2000

[all]
dtoverlay=
# the SD card is polled once, at boot: without it the power off of the
# uConsole may take a long time
dtparam=sd_poll_once
dtparam=ant2

dtoverlay=axp22x-soft-battery
dtparam=v_off=2700000
dtparam=energy_full_design_uwh=32332474

# The curves of this battery, worked out from its runs; no package installs
# this overlay, the firmware skips it while it is not there. Make it:
#   1. systemctl enable --now axp22x-soft-battery.service, then charge
#      the battery full and run it down to the power off, once or more
#      (the runs go to /var/lib/axp22x-soft-battery);
#   2. axp22x-soft-battery gen-dts > /tmp/axp22x-soft-battery-ocv.dts
#      (--use-partial takes also a run cut at 10 % or lower);
#   3. dtc -@ -I dts -O dtb -o /boot/overlays/axp22x-soft-battery-ocv.dtbo \
#          /tmp/axp22x-soft-battery-ocv.dts
#   4. reboot.
dtoverlay=axp22x-soft-battery-ocv

ignore_lcd=1
max_framebuffers=2
disable_overscan=1
dtoverlay=dwc2,dr_mode=host
# the headphone jack (high: headphones in), the amplifier of the speakers
# off until the driver of the sound card takes it (the overlay of the CM5)
gpio=10=ip,np
gpio=11=op,dl
gpio=9=op,dh

# The kernel with 16K pages (sys-kernel/uconsole-pi-kernel, USE 16k), for
# one boot only: sudo reboot '0 tryboot'. The boot after it takes
# kernel_2712.img again. To keep it, put the line under [pi5] above.
[pi5]
[tryboot]
kernel=kernel_2712_16k.img
[all]
